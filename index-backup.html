    let gameActive=false,score=0,lives=3,level=1,enemySpeed=1.4,enemySpawnRate=1900,playerX=200,playerY=200,playerElement=null,playArea=null;
    let lastLevel=1;
    let obstaclesArray=[],obstacleInterval,enemiesArray=[],tearsArray=[],notesArray=[],powerupsArray=[];
    let enemyInterval,tearInterval,noteInterval,powerupInterval,gameLoopInterval,activePowerup=null,powerupTimer=null,areaWidth=0,areaHeight=0;
    const POWERUP_TYPES={SHOWER:"💧💥 SHOWER POWER",CRYBURST:"😭💢 CRY BURST",MAXIMO:"👑 MAXIMO MODE"};
    function showScreen(id){document.querySelectorAll(".screen").forEach(s=>s.classList.remove("active"));document.getElementById(id).classList.add("active")}
    function updateUI(){if(!gameActive)return;level=Math.floor(score/200)+1;enemySpeed=1.4+(level-1)*0.3;document.getElementById("scoreValue").innerText=score;document.getElementById("livesValue").innerText=lives;if(activePowerup){let s=Math.ceil((activePowerup.endTime-Date.now())/1000);document.getElementById("powerupStatus").innerHTML=s>0?`✨ ${activePowerup.type} ${s}s`:"⚡ NORMAL"}else{document.getElementById("powerupStatus").innerHTML="⚡ NORMAL"}}
    function collide(a,b){let r1=a.getBoundingClientRect(),r2=b.getBoundingClientRect();return !(r1.top>r2.bottom||r1.bottom<r2.top||r1.right<r2.left||r1.left>r2.right)}
    function restorePlayerImage(){if(!playerElement)return;playerElement.style.backgroundImage='url("reh-dogg.png")';playerElement.style.backgroundSize="contain";playerElement.style.backgroundRepeat="no-repeat";playerElement.style.backgroundPosition="center";playerElement.style.boxShadow="";playerElement.style.filter=""}
    function removePowerupVisuals(){restorePlayerImage();document.getElementById("powerupStatus").style.background="#2c0a0a"}
    function applyPowerupEffect(type){if(type===POWERUP_TYPES.CRYBURST){enemiesArray.forEach(e=>e?.remove());enemiesArray=[];let b=document.createElement("div");b.style.position="absolute";b.style.inset="0";b.style.background="radial-gradient(circle,rgba(255,100,100,.6),transparent)";b.style.pointerEvents="none";b.style.zIndex="100";playArea.appendChild(b);setTimeout(()=>b.remove(),400)}if(type===POWERUP_TYPES.MAXIMO&&playerElement){playerElement.style.boxShadow="0 0 26px gold";playerElement.style.filter="drop-shadow(0 0 18px gold)"}if(type===POWERUP_TYPES.SHOWER&&playerElement){playerElement.style.filter="drop-shadow(0 0 8px cyan)"}}
    function spawnPowerup(){if(!gameActive)return;let emojis=["🚿💧","😭💢","👑⭐"],types=[POWERUP_TYPES.SHOWER,POWERUP_TYPES.CRYBURST,POWERUP_TYPES.MAXIMO],idx=Math.floor(Math.random()*types.length);let d=document.createElement("div");d.className="powerup";d.innerText=emojis[idx];d.style.left=Math.random()*(areaWidth-50)+20+"px";d.style.top=Math.random()*(areaHeight-50)+20+"px";d.dataset.type=types[idx];playArea.appendChild(d);powerupsArray.push(d);setTimeout(()=>{if(d?.remove){let i=powerupsArray.indexOf(d);if(i!==-1)powerupsArray.splice(i,1);d.remove()}},7000)}
    function checkPowerupCollision(){for(let i=0;i<powerupsArray.length;i++){let p=powerupsArray[i];if(p&&playerElement&&collide(playerElement,p)){let t=p.dataset.type;if(activePowerup){if(powerupTimer)clearTimeout(powerupTimer);removePowerupVisuals()}activePowerup={type:t,endTime:Date.now()+7000};updateUI();applyPowerupEffect(t);if(powerupTimer)clearTimeout(powerupTimer);powerupTimer=setTimeout(()=>{if(activePowerup&&activePowerup.type===t){activePowerup=null;removePowerupVisuals();updateUI()}},7000);p.remove();powerupsArray.splice(i,1);break}}}
    function playerHit(){lives--;updateUI();if(playerElement){playerElement.classList.add("player-hit");setTimeout(()=>playerElement?.classList.remove("player-hit"),350)}if(lives<=0){endGame();return false}return true}
    function handleEnemyCollisions(){for(let i=0;i<enemiesArray.length;i++){let e=enemiesArray[i];if(e&&playerElement&&collide(playerElement,e)){let inv=activePowerup&&activePowerup.type===POWERUP_TYPES.MAXIMO;e.remove();enemiesArray.splice(i,1);if(inv){playerElement.style.opacity=".7";setTimeout(()=>{if(playerElement)playerElement.style.opacity="1"},180);return true}return playerHit()}}return true}
    function handleObstacleCollisions(){for(let i=0;i<obstaclesArray.length;i++){let o=obstaclesArray[i];if(o&&playerElement&&collide(playerElement,o)){let inv=activePowerup&&activePowerup.type===POWERUP_TYPES.MAXIMO;o.remove();obstaclesArray.splice(i,1);if(inv)return true;return playerHit()}}return true}
    function showScorePop(amount){if(!playerElement||!playArea)return;let pop=document.createElement("div");pop.className="score-pop";pop.innerText="+"+amount;pop.style.left=(playerX+20)+"px";pop.style.top=(playerY)+"px";playArea.appendChild(pop);setTimeout(()=>{if(pop?.remove)pop.remove()},900)}
    function collectItems(){let m=(activePowerup&&activePowerup.type===POWERUP_TYPES.SHOWER)?2:1;for(let i=0;i<tearsArray.length;i++){let t=tearsArray[i];if(t&&playerElement&&collide(playerElement,t)){let gain=10*m;score+=gain;showScorePop(gain);t.remove();tearsArray.splice(i,1);updateUI();i--}}for(let i=0;i<notesArray.length;i++){let n=notesArray[i];if(n&&playerElement&&collide(playerElement,n)){let gain=25*m;score+=gain;showScorePop(gain);n.remove();notesArray.splice(i,1);updateUI();i--}}}
    function spawnTearTrail(){if(!playerElement||!gameActive)return;let t=document.createElement("div");t.className="tear-trail";t.innerText="💧";t.style.left=(playerX+18)+"px";t.style.top=(playerY+40)+"px";playArea.appendChild(t);setTimeout(()=>t?.remove(),800)}
    function movePlayer(dx,dy){if(!gameActive||!playerElement)return;playerX=Math.max(12,Math.min(areaWidth-65,playerX+dx));playerY=Math.max(12,Math.min(areaHeight-100,playerY+dy));playerElement.style.left=playerX+"px";playerElement.style.top=playerY+"px";spawnTearTrail();checkPowerupCollision();collectItems();if(!handleEnemyCollisions())return;if(!handleObstacleCollisions())return;if(lives<=0)endGame()}
    function move(dir){let step=26;if(dir==="up")movePlayer(0,-step);if(dir==="down")movePlayer(0,step);if(dir==="left")movePlayer(-step,0);if(dir==="right")movePlayer(step,0)}
    function spawnEnemy(){if(!gameActive)return;let e=document.createElement("div");e.className="enemy";e.style.left=Math.random()*(areaWidth-70)+10+"px";e.style.top=Math.random()*(areaHeight-70)+10+"px";playArea.appendChild(e);enemiesArray.push(e);setTimeout(()=>{if(e?.remove){let i=enemiesArray.indexOf(e);if(i!==-1)enemiesArray.splice(i,1);e.remove()}},10000)}
    function spawnObstacle(){if(!gameActive)return;let types=["🌲","🪨","🪵"];let o=document.createElement("div");o.className="obstacle";o.textContent=types[Math.floor(Math.random()*types.length)];o.style.left=Math.random()*(areaWidth-70)+10+"px";o.style.top=Math.random()*(areaHeight-70)+10+"px";playArea.appendChild(o);obstaclesArray.push(o);setTimeout(()=>{if(o?.remove){let i=obstaclesArray.indexOf(o);if(i!==-1)obstaclesArray.splice(i,1);o.remove()}},10000)}
    function spawnTear(){if(!gameActive)return;let t=document.createElement("div");t.className="tear";t.innerText="💧";t.style.left=Math.random()*(areaWidth-50)+10+"px";t.style.top=Math.random()*(areaHeight-50)+10+"px";playArea.appendChild(t);tearsArray.push(t);setTimeout(()=>{if(t?.remove){let i=tearsArray.indexOf(t);if(i!==-1)tearsArray.splice(i,1);t.remove()}},8000)}
    function spawnNote(){if(!gameActive)return;let n=document.createElement("div");n.className="note";n.innerText="🎵";n.style.left=Math.random()*(areaWidth-50)+10+"px";n.style.top=Math.random()*(areaHeight-50)+10+"px";playArea.appendChild(n);notesArray.push(n);setTimeout(()=>{if(n?.remove){let i=notesArray.indexOf(n);if(i!==-1)notesArray.splice(i,1);n.remove()}},8000)}
    function moveEnemiesTowardPlayer(){if(!gameActive||!playerElement)return;enemiesArray.forEach(e=>{if(!e)return;let ex=parseFloat(e.style.left)||0,ey=parseFloat(e.style.top)||0,dx=playerX-ex,dy=playerY-ey,d=Math.sqrt(dx*dx+dy*dy);if(d>0){ex+=(dx/d)*enemySpeed;ey+=(dy/d)*enemySpeed;e.style.left=ex+"px";e.style.top=ey+"px"}})}
    function clearGameIntervals(){if(enemyInterval)clearInterval(enemyInterval);if(tearInterval)clearInterval(tearInterval);if(noteInterval)clearInterval(noteInterval);if(powerupInterval)clearInterval(powerupInterval);if(obstacleInterval)clearInterval(obstacleInterval);if(gameLoopInterval)clearInterval(gameLoopInterval);if(powerupTimer)clearTimeout(powerupTimer)}
    function resetGameVariables(){score=0;lives=3;enemiesArray.forEach(e=>e?.remove());tearsArray.forEach(t=>t?.remove());notesArray.forEach(n=>n?.remove());powerupsArray.forEach(p=>p?.remove());obstaclesArray.forEach(o=>o?.remove());enemiesArray=[];tearsArray=[];notesArray=[];powerupsArray=[];obstaclesArray=[];activePowerup=null;if(powerupTimer)clearTimeout(powerupTimer);restorePlayerImage()}
